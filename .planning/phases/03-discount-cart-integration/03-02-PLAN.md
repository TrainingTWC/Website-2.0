  <action>
    **1. Add activeDiscount state** — immediately after the existing `useState(false)` for `cartOpen`, add:

    ```typescript
    const [activeDiscount, setActiveDiscount] = useState<{
      code: string;
      discountType: "percent" | "flat";
      amount: number;
    } | null>(() => {
      try {
        const raw = localStorage.getItem("twc_active_discount");
        return raw ? JSON.parse(raw) : null;
      } catch {
        return null;
      }
    });
    ```

    **2. Add clearDiscount callback** — alongside `addToCart`, `removeFromCart`, `updateQty`:

    ```typescript
    const clearDiscount = useCallback(() => {
      setActiveDiscount(null);
      localStorage.removeItem("twc_active_discount");
    }, []);
    ```

    **3. Add discountedSubtotal memo** — after `cartCount`:

    ```typescript
    const rawSubtotal = useMemo(
      () =>
        cart.reduce((sum, item) => {
          const p = (products ?? []).find((pr) => pr._id === item.productId);
          return sum + (p?.price ?? 0) * item.qty;
        }, 0),
      [cart, products]
    );

    const discountedSubtotal = useMemo(() => {
      if (!activeDiscount) return rawSubtotal;
      if (activeDiscount.discountType === "percent") {
        return Math.round(rawSubtotal * (1 - activeDiscount.amount / 100));
      }
      return Math.max(0, rawSubtotal - activeDiscount.amount);
    }, [rawSubtotal, activeDiscount]);
    ```

    Note: CartPanel already computes its own `subtotal` internally from props — `discountedSubtotal` passed via prop is used only for display. `rawSubtotal` here matches CartPanel's internal subtotal.
