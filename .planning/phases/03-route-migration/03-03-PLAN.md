navigateTo() → router.push() mapping (ALL occurrences in App.tsx to replace in HomeContent.tsx):
```
navigateTo({ page: null })                    → router.push("/")
navigateTo({ page: null, product: null, ... }) → router.push("/")
navigateTo({ page: "shop" })                   → router.push("/shop")
navigateTo({ page: "third-circle" })           → router.push("/journal")
navigateTo({ page: "third-circle", post: null })→ router.push("/journal")
navigateTo({ page: "checkout" })               → router.push("/checkout")
navigateTo({ page: "order-portal" })           → router.push("/orders")
navigateTo({ page: "ti" })                     → setTiOpen(true)  ← TI stays as overlay state
navigateTo({ product: slug })                  → router.push("/products/" + slug)
navigateTo({ product: slugify(p.name) })       → router.push("/products/" + p._id)  ← ROUTE-03: use _id not slug
```
